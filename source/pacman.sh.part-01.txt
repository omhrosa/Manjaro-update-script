        case "$cmd" in
            sudo) MISSING_REPO_PKGS+=(sudo) ;;
            pacman | pacman-key) MISSING_REPO_PKGS+=(pacman) ;;
            smartctl) MISSING_REPO_PKGS+=(smartmontools) ;;
            snapper) MISSING_REPO_PKGS+=(snapper) ;;
            grub-mkconfig) MISSING_REPO_PKGS+=(grub) ;;
            pacman-mirrors) MISSING_REPO_PKGS+=(pacman-mirrors) ;;
            curl) MISSING_REPO_PKGS+=(curl) ;;
            flatpak) MISSING_REPO_PKGS+=(flatpak) ;;
            yay) MISSING_REPO_PKGS+=(yay) ;;
            gext) MISSING_AUR_PKGS+=(gnome-extensions-cli) ;;
            checkrebuild) MISSING_AUR_PKGS+=(rebuild-detector) ;;
            fuser) MISSING_REPO_PKGS+=(psmisc) ;;
            python3) MISSING_REPO_PKGS+=(python) ;;
            timedatectl | systemctl) MISSING_REPO_PKGS+=(systemd) ;;
            findmnt | lsblk | script | flock) MISSING_REPO_PKGS+=(util-linux) ;;
            mhwd-kernel) MISSING_REPO_PKGS+=(mhwd) ;;
            xdg-open) MISSING_REPO_PKGS+=(xdg-utils) ;;
            meld) MISSING_REPO_PKGS+=(meld) ;;
            gnome-text-editor) MISSING_REPO_PKGS+=(gnome-text-editor) ;;
            awk) MISSING_REPO_PKGS+=(gawk) ;;
            sed) MISSING_REPO_PKGS+=(sed) ;;
            grep) MISSING_REPO_PKGS+=(grep) ;;
            find) MISSING_REPO_PKGS+=(findutils) ;;
            ldd) MISSING_REPO_PKGS+=(glibc) ;;
            realpath | stat | df | tr | sort | comm | cmp | head | wc | tail | cut | basename | dirname | date | mkdir | rm | cp | mv | cat | touch | chmod | chown | tee | mktemp | stty | sleep | yes | ls | uname)
                MISSING_REPO_PKGS+=(coreutils)
                ;;
        esac
    done

    local -a uniq_repo=() uniq_aur=()
    for p in "${MISSING_REPO_PKGS[@]}"; do
        [[ " ${uniq_repo[*]} " == *" $p "* ]] || uniq_repo+=("$p")
    done
    for p in "${MISSING_AUR_PKGS[@]}"; do
        [[ " ${uniq_aur[*]} " == *" $p "* ]] || uniq_aur+=("$p")
    done
    MISSING_REPO_PKGS=("${uniq_repo[@]}")
    MISSING_AUR_PKGS=("${uniq_aur[@]}")
}

# Commands required to complete SMART/Snapper preflight cannot be auto-installed
# before those safety gates without defeating the preflight policy.
preflight_dependencies_ready() {
    local -a required=(smartctl snapper grub-mkconfig python3 timedatectl systemctl findmnt lsblk realpath awk sort head date)
    local -a missing=()
    local cmd
    for cmd in "${required[@]}"; do
        command -v "$cmd" > /dev/null 2>&1 || missing+=("$cmd")
    done
    if ((${#missing[@]} > 0)); then
        echo -e "${red}Missing preflight dependencies:${reset} ${missing[*]}"
        echo -e "${yellow}They cannot be auto-installed before SMART/Snapper checks. Install them and rerun.${reset}"
        return 1
    fi
    return 0
}

install_detected_dependencies() {
    ((${#MISSING_CMDS[@]} == 0)) && return 0

    local ans p
    echo
    echo -e "${yellow}Missing dependencies detected.${reset}"
    echo -e "${orange}Missing commands:${reset} ${MISSING_CMDS[*]}"
    ((${#MISSING_REPO_PKGS[@]})) && echo -e "${orange}Repo packages:${reset} ${MISSING_REPO_PKGS[*]}"
    ((${#MISSING_AUR_PKGS[@]})) && echo -e "${orange}AUR packages:${reset} ${MISSING_AUR_PKGS[*]}"
    echo -ne "${yellow}Install all missing dependencies now? [y/N]: ${reset}"
    read -r ans
    [[ "${ans,,}" == "y" || "${ans,,}" == "yes" ]] || return 1

    # Keep the existing defensive forced database refresh.
    sudo pacman -Syy --noconfirm || return 1

    local -a repo_pkgs=("${MISSING_REPO_PKGS[@]}")
    if ((NEED_YAY)); then
        sudo pacman -S --needed --noconfirm yay || return 1
        local -a tmp=()
        for p in "${repo_pkgs[@]}"; do
            [[ "$p" == "yay" ]] || tmp+=("$p")
        done
        repo_pkgs=("${tmp[@]}")
    fi

    if ((${#repo_pkgs[@]})); then
        sudo pacman -S --needed --noconfirm "${repo_pkgs[@]}" || return 1
    fi
    if ((${#MISSING_AUR_PKGS[@]})); then
        yay -S --needed --noconfirm "${MISSING_AUR_PKGS[@]}" || return 1
    fi

    # Confirm that the dependency pass actually resolved everything.
    detect_dependencies
    if ((${#MISSING_CMDS[@]} > 0)); then
        echo -e "${red}Dependencies still missing after installation:${reset} ${MISSING_CMDS[*]}"
        return 1
    fi
    return 0
}

    pk="$(lsblk -no PKNAME "$disk" 2> /dev/null | head -n 1)"
    [[ -z "$pk" ]] && break
    disk="/dev/$pk"

    echo -e "\n${red}Error: could not detect OS disk device for /.${reset}"
    echo -e "\n${red}Exiting script...${reset}"
